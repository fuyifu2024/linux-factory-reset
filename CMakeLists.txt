cmake_minimum_required(VERSION 3.20)
project(linux-factory-reset VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_INSTALL_DEFAULT_DIRECTORY_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
find_package(CURL REQUIRED)
find_package(nlohmann_json 3.2 REQUIRED)
add_library(reset_core src/core.cpp src/system.cpp src/download.cpp)
target_include_directories(reset_core PUBLIC include)
target_link_libraries(reset_core PUBLIC OpenSSL::Crypto CURL::libcurl nlohmann_json::nlohmann_json)
target_compile_options(reset_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_executable(linux-factory-reset src/main.cpp src/gui.cpp)
target_link_libraries(linux-factory-reset PRIVATE reset_core)
target_compile_options(linux-factory-reset PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
include(CTest)
if(BUILD_TESTING)
    add_executable(test-core tests/core.cpp)
    target_link_libraries(test-core PRIVATE reset_core)
    target_compile_definitions(test-core PRIVATE UFRESET_TEST_ASSETS="${CMAKE_SOURCE_DIR}/firstboot")
    target_compile_definitions(test-core PRIVATE UFRESET_TEST_APP="$<TARGET_FILE:linux-factory-reset>")
    add_dependencies(test-core linux-factory-reset)
    add_executable(test-download tests/download.cpp)
    target_link_libraries(test-download PRIVATE reset_core)
    add_test(NAME core COMMAND test-core)
    add_test(NAME reconcile COMMAND sh ${CMAKE_SOURCE_DIR}/tests/reconcile.sh ${CMAKE_SOURCE_DIR}/boot/reconcile.sh)
    add_test(NAME relocate COMMAND sh ${CMAKE_SOURCE_DIR}/tests/relocate.sh ${CMAKE_SOURCE_DIR}/boot/relocate.sh)
    add_test(NAME cli-help COMMAND linux-factory-reset --help)
    add_test(NAME gui-preview COMMAND linux-factory-reset gui --preview)
    add_test(NAME boot-syntax COMMAND sh -n ${CMAKE_SOURCE_DIR}/boot/local-bottom)
endif()
install(TARGETS linux-factory-reset RUNTIME DESTINATION bin)
install(PROGRAMS packaging/ubuntu-reset DESTINATION bin)
install(PROGRAMS boot/hook DESTINATION /etc/initramfs-tools/hooks RENAME ubuntu-reset)
install(PROGRAMS boot/local-bottom DESTINATION /etc/initramfs-tools/scripts/local-bottom RENAME ubuntu-reset)
install(FILES boot/reconcile.sh DESTINATION lib/ubuntu-reset)
install(FILES boot/relocate.sh DESTINATION lib/ubuntu-reset)
install(PROGRAMS firstboot/firstboot.sh DESTINATION lib/ubuntu-reset)
install(FILES firstboot/ubuntu-reset-firstboot.service DESTINATION lib/ubuntu-reset)
install(FILES packaging/linux-factory-reset.desktop DESTINATION share/applications)
install(FILES packaging/linux-factory-reset.svg DESTINATION share/icons/hicolor/scalable/apps)
install(FILES LICENSE README.md README.en.md DESTINATION share/doc/linux-factory-reset)
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_CONTACT "fuyifu2024 <fuyifu2024@users.noreply.github.com>")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Experimental Linux recovery preparation and official image downloader")
set(CPACK_PACKAGE_VERSION "0.4.0~alpha1")
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_STRIP_FILES TRUE)
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS TRUE)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "curl, gpgv, rsync, squashfs-tools, xz-utils, util-linux, initramfs-tools, zenity, pkexec, udev, findutils, qemu-utils")
# Resolve libssl3/libssl3t64 and other libraries from the actual distro build.
execute_process(COMMAND dpkg-vendor --query Vendor OUTPUT_VARIABLE RESET_VENDOR OUTPUT_STRIP_TRAILING_WHITESPACE)
if(RESET_VENDOR STREQUAL "Ubuntu")
    set(CPACK_DEBIAN_PACKAGE_RECOMMENDS "ubuntu-keyring")
endif()
set(CPACK_DEBIAN_PACKAGE_PROVIDES "ubuntu-reset")
set(CPACK_DEBIAN_PACKAGE_REPLACES "ubuntu-reset")
set(CPACK_DEBIAN_PACKAGE_CONFLICTS "ubuntu-reset")
include(CPack)
